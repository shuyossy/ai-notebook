import EmbeddedPostgres from 'embedded-postgres';
import { existsSync } from 'node:fs';

const dataDir = './pgdata';
const pg = new EmbeddedPostgres({
  databaseDir: dataDir,
  user: 'postgres',
  password: 'postgres-pass',
  port: 5432,
  persistent: true,
  createPostgresUser: process.env.AS_ROOT === '1',
});

if (!existsSync(`${dataDir}/PG_VERSION`)) await pg.initialise();
await pg.start();
console.log('PostgreSQL started on localhost:5432');

const stop = async () => { await pg.stop(); process.exit(0); };
process.on('SIGINT', stop);
process.on('SIGTERM', stop);

import EmbeddedPostgres from 'embedded-postgres';

const [db, sql] = process.argv.slice(2);
const client = new EmbeddedPostgres({
  user: process.env.DBUSER ?? 'postgres',
  password: process.env.DBPASS ?? 'postgres-pass',
  port: 5432,
}).getPgClient(db);

await client.connect();
const res = await client.query(sql);
if (res.rows?.length) console.table(res.rows); else console.log(res.command, 'OK');
await client.end();
